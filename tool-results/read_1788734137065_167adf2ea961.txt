     9→/**
    10→ * Alert threshold evaluation engine (Task 5-a).
    11→ *
    12→ * Pure-Next implementation so it is reusable and reviewable: the worker
    13→ * never opens SQLite — it claims an ALERT_EVALUATION job and calls
    14→ * POST /api/v1/alerts/evaluate, which runs `runAlertEvaluation` below
    15→ * (evaluate-in-Next, same architecture as the 3-c drift engine).
    16→ *
    17→ * Evaluation pipeline (documented choices):
    18→ *   1. Load every ACTIVE AlertRule; resolve scopeJson → device set.
    19→ *      Scope keys follow the 3-a conventions ({ siteCodes, criticalities,
    20→ *      deviceRoles }); UNMANAGED devices are always excluded; OFFLINE
    21→ *      devices are excluded for normal metrics but evaluated for the
    22→ *      AVAILABILITY pseudo-metric (they are exactly its target).
    23→ *   2. Threshold metrics (CPU, MEMORY, UTILIZATION_IN/OUT, LATENCY_MS,
    24→ *      PACKET_LOSS, SESSIONS, TEMPERATURE): the rule breaches when the AVG
    25→ *      of MetricSamples inside the rule's durationMinutes window satisfies
    26→ *      the operator/threshold (avg-over-window — smooths the 5-min sample
    27→ *      spikes, documented instead of "latest"). No samples in the window ⇒
    28→ *      no fire and no resolve (sparse data never flaps the stream).
    29→ *      Pseudo-metric AVAILABILITY: 0 when the device is OFFLINE or UNKNOWN
    30→ *      with lastSeen older than 10 min, else 1 — state-based, no samples.
    31→ *   3. Dedup: fingerprint dedupKey = "<deviceId>:<metric>:<ruleId>". An
    32→ *      existing ACTIVE/ACKNOWLEDGED alert with the same key is updated
    33→ *      (lastSeen + count) instead of duplicated. Legacy rows (ruleId set,
    34→ *      dedupKey null — Phase-1 seed) are matched by deviceId+ruleId and
    35→ *      migrated onto the fingerprint on first touch.
    36→ *   4. Maintenance-window suppression: a device (or its site) inside an
    37→ *      active MaintenanceWindow never fires — new alerts are born
    38→ *      SUPPRESSED with suppressReason "Maintenance window: <name>", and
    39→ *      existing ACTIVE engine alerts are suppressed the same way
    40→ *      (ACKNOWLEDGED ones keep the human ownership and stay).
    41→ *   5. Root/child grouping (Gate G5): when a device is unreachable, the
    42→ *      AVAILABILITY rule produces the ROOT alert; any other rule breaching
    43→ *      for the same device (and any existing ACTIVE engine alert on it)
    44→ *      becomes/turns into a SUPPRESSED CHILD (parentAlertId set,
    45→ *      suppressReason "Suppressed by root alert"). The root fires alone.
    46→ *      While the root is open its children stay suppressed; after the root
    47→ *      resolves, children re-activate if their own condition still holds,
    48→ *      or resolve with it.
    49→ *   6. Auto-resolve: a rule that no longer breaches resolves its open
    50→ *      alert — threshold metrics need the recent window AND the previous
    51→ *      equal window both non-breaching (2 consecutive windows); recovery
    52→ *      of AVAILABILITY resolves immediately (state-based).
    53→ *   7. Incident auto-creation: a NEW CRITICAL/HIGH alert (not a dedup
    54→ *      update, not suppressed) on a device WITHOUT an open incident
    55→ *      creates one through the shared module (src/lib/incidents/create.ts)
    56→ *      — SEV1..SEV4 mapping + SLA due — and links Alert.incidentId.
    57→ *      Incidents are human-resolved (5-b); the engine never closes them.
    58→ *   8. Notifications: broadcast rows for CRITICAL/HIGH fires and for
    59→ *      auto-created incidents (§74 surface, capped per run).
    60→ *   9. Audit: ALERT_FIRED / ALERT_SUPPRESSED / ALERT_RESOLVED per alert
    61→ *      plus one ALERT_EVALUATION_COMPLETED summary per run
    62→ *      (actor "system:alert-engine").
    63→ *
    64→ * Flood caps per run: 50 new alerts, 10 incidents, 40 notifications.
    65→ */

   158→function compare(a: number, operator: string, b: number): boolean {
   159→  switch (operator) {
   160→    case "GT":
   161→      return a > b;
   162→    case "GTE":
   163→      return a >= b;
   164→    case "LT":
   165→      return a < b;
   166→    case "LTE":
   167→      return a <= b;
   168→    case "EQ":
   169→      return Math.abs(a - b) < 1e-9;
   170→    default:
   171→      return false;
   172→  }
   173→}
   174→
   175→function fmt(value: number): string {
   176→  return Number.isInteger(value) ? String(value) : value.toFixed(1);
   177→}

   179→/** Pseudo-metric availability: 0 unreachable, 1 reachable. */
   180→function availabilityOf(device: EvalDevice, now: Date): number {
   181→  if (device.status === "OFFLINE") return 0;
   182→  if (
   183→    device.status === "UNKNOWN" &&
   184→    (!device.lastSeen ||
   185→      now.getTime() - device.lastSeen.getTime() >
   186→        AVAILABILITY_STALE_UNKNOWN_MIN * 60_000)
   187→  ) {
   188→    return 0;
   189→  }
   190→  return 1;
   191→}

   193→/**
   194→ * Device set for one rule. Reuses the 3-a scope parsing conventions
   195→ * (siteCodes/criticalities via parsePolicyScope) and adds deviceRoles.
   196→ * OFFLINE devices are excluded for everything except AVAILABILITY.
   197→ */
   198→async function ruleDeviceWhere(
   199→  rule: EvalRule,
   200→  scopeText: string | null | undefined
   201→) {
   202→  const scope = parsePolicyScope(scopeText);
   203→  const roles =
   204→    scopeText
   205→      ? (() => {
   206→          try {
   207→            const parsed: unknown = JSON.parse(scopeText);
   208→            const raw =
   209→              parsed && typeof parsed === "object" && !Array.isArray(parsed)
   210→                ? (parsed as Record<string, unknown>).deviceRoles
   211→                : null;
   212→            return Array.isArray(raw)
   213→              ? raw.filter((r): r is string => typeof r === "string" && r.length > 0)
   214→              : null;
   215→          } catch {
   216→            return null;
   217→          }
   218→        })()
   219→      : null;
   220→
   221→  const statusExclusions: string[] = ["UNMANAGED"];
   222→  if (rule.metric !== "AVAILABILITY") statusExclusions.push("OFFLINE");
   223→
   224→  return {
   225→    AND: [
   226→      { status: { notIn: statusExclusions } },
   227→      ...(scope.criticalities
   228→        ? [{ criticality: { in: scope.criticalities } }]
   229→        : []),
   230→      ...(scope.siteCodes && !scope.siteCodes.includes("*")
   231→        ? [{ site: { code: { in: scope.siteCodes } } }]
   232→        : []),
   233→      ...(roles ? [{ role: { in: roles } }] : []),
   234→    ],
   235→  };
   236→}

   238→/**
   239→ * Run one full evaluation pass. `correlationId` ties audits together
   240→ * (engine passes the job's correlation id; manual runs get ALR-XXXXXX).
   241→ */
   242→export async function runAlertEvaluation(options: {
   243→  triggeredBy?: string;
   244→  correlationId?: string;
   245→}): Promise<AlertEvaluationSummary> {
   246→  const now = new Date();
   247→  const triggeredBy = options.triggeredBy ?? "MANUAL";
   248→  const correlationId = options.correlationId ?? newCorrelationId("ALR");
   249→
   250→  const summary: AlertEvaluationSummary = {
   251→    evaluatedAt: now.toISOString(),
   252→    triggeredBy,
   253→    rulesEvaluated: 0,
   254→    devicesConsidered: 0,
   255→    fired: 0,
   256→    deduped: 0,
   257→    suppressed: 0,
   258→    childrenSuppressed: 0,
   259→    resolved: 0,
   260→    incidentsCreated: 0,
   261→    notificationsCreated: 0,
   262→    caps: { newAlerts: false, incidents: false, notifications: false },
   263→  };
   264→
   265→  const rules = await db.alertRule.findMany({ where: { isActive: true } });
   266→  if (rules.length === 0) {
   267→    return summary;
   268→  }
   269→  summary.rulesEvaluated = rules.length;
   270→
   271→  /* ── resolve scopes → device sets ──────────────────────────────────── */
   272→  const deviceById = new Map<string, EvalDevice>();
   273→  const pairs: Array<{ rule: EvalRule; device: EvalDevice }> = [];
   274→  for (const rule of rules) {
   275→    const where = await ruleDeviceWhere(rule, rule.scopeJson);
   276→    const devices = await db.device.findMany({
   277→      where,
   278→      select: {
   279→        id: true,
   280→        hostname: true,
   281→        status: true,
   282→        lastSeen: true,
   283→        siteId: true,
   284→      },
   285→    });
   286→    for (const device of devices) {
   287→      deviceById.set(device.id, device);
   288→      pairs.push({ rule, device });
   289→    }
   290→  }
   291→  summary.devicesConsidered = deviceById.size;
   292→  if (pairs.length === 0) return summary;
   293→
   294→  // AVAILABILITY (root) rules evaluate FIRST so dependent alerts fired in
   295→  // the same pass are parented to the real root row, never a placeholder.
   296→  pairs.sort(
   297→    (a, b) =>
   298→      (a.rule.metric === "AVAILABILITY" ? 0 : 1) -
   299→      (b.rule.metric === "AVAILABILITY" ? 0 : 1)
   300→  );
   301→
   302→  /* ── maintenance windows (device-scoped + site-scoped) ─────────────── */
   303→  const windows = await db.maintenanceWindow.findMany({
   304→    where: {
   305→      isActive: true,
   306→      startsAt: { lte: now },
   307→      endsAt: { gte: now },
   308→    },
   309→    select: { id: true, name: true, deviceId: true, siteId: true },
   310→  });
   311→  const windowByDevice = new Map<string, string>();
   312→  const windowBySite = new Map<string, string>();
   313→  for (const w of windows) {
   314→    if (w.deviceId && !windowByDevice.has(w.deviceId)) {
   315→      windowByDevice.set(w.deviceId, w.name);
   316→    }
   317→    if (w.siteId && !windowBySite.has(w.siteId)) {
   318→      windowBySite.set(w.siteId, w.name);
   319→    }
   320→  }
   321→  const maintenanceFor = (device: EvalDevice): string | null =>
   322→    windowByDevice.get(device.id) ??
   323→    (device.siteId ? windowBySite.get(device.siteId) : undefined) ??
   324→    null;
   325→
   326→  /* ── existing engine-managed alerts ────────────────────────────────── */
   327→  const existingRows = await db.alert.findMany({
   328→    where: {
   329→      status: { in: [...OPEN_STATUSES] },
   330→      OR: [
   331→        { dedupKey: { not: null } },
   332→        { ruleId: { not: null }, dedupKey: null },
   333→      ],
   334→    },
   335→    select: {
   336→      id: true,
   337→      deviceId: true,
   338→      ruleId: true,
   339→      dedupKey: true,
   340→      status: true,
   341→      suppressReason: true,
   342→      parentAlertId: true,
   343→    },
   344→  });
   345→  const existingByKey = new Map<string, ExistingAlert>();
   346→  const existingByDeviceRule = new Map<string, ExistingAlert>();
   347→  for (const row of existingRows) {
   348→    if (row.dedupKey && !existingByKey.has(row.dedupKey)) {
   349→      existingByKey.set(row.dedupKey, row);
   350→    }
   351→    if (row.ruleId && !row.dedupKey) {
   352→      const key = `${row.deviceId}:${row.ruleId}`;
   353→      if (!existingByDeviceRule.has(key)) existingByDeviceRule.set(key, row);
   354→    }
   355→  }
   356→  /** Root alert (AVAILABILITY) currently open per device — built as we go. */
   357→  const openRootByDevice = new Map<string, ExistingAlert>();
   358→  for (const row of existingRows) {
   359→    if (
   360→      row.dedupKey?.includes(":AVAILABILITY:") &&
   361→      row.status === "ACTIVE" &&
   362→      !openRootByDevice.has(row.deviceId)
   363→    ) {
   364→      openRootByDevice.set(row.deviceId, row);
   365→    }
   366→  }
   367→
   368→  /* ── load samples for every needed (device, metric) combo ──────────── */
   369→  const neededMetrics = new Set(
   370→    rules.filter((r) => r.metric !== "AVAILABILITY").map((r) => r.metric)
   371→  );
   372→  const maxWindowMin = Math.max(
   373→    ...rules.filter((r) => r.metric !== "AVAILABILITY").map((r) => r.durationMinutes),
   374→    0
   375→  );
   376→  const samplesByDeviceMetric = new Map<string, Array<{ ts: Date; value: number }>>();
   377→  if (neededMetrics.size > 0 && maxWindowMin > 0) {
   378→    // Resolve pass needs the previous window too → load 2× the max window.
   379→    const since = new Date(now.getTime() - 2 * maxWindowMin * 60_000);
   380→    const samples = await db.metricSample.findMany({
   381→      where: {
   382→        deviceId: { in: [...deviceById.keys()] },
   383→        metric: { in: [...neededMetrics] },
   384→        ts: { gte: since },
   385→      },
   386→      select: { deviceId: true, metric: true, value: true, ts: true },
   387→      orderBy: { ts: "asc" },
   388→    });
   389→    for (const s of samples) {
   390→      const key = `${s.deviceId}:${s.metric}`;
   391→      const list = samplesByDeviceMetric.get(key);
   392→      if (list) list.push({ ts: s.ts, value: s.value });
   393→      else samplesByDeviceMetric.set(key, [{ ts: s.ts, value: s.value }]);
   394→    }
   395→  }
   396→
   397→  const windowAvg = (
   398→    deviceId: string,
   399→    metric: string,
   400→    fromMs: number,
   401→    toMs: number
   402→  ): number | null => {
   403→    const list = samplesByDeviceMetric.get(`${deviceId}:${metric}`) ?? [];
   404→    const inWindow = list.filter(
   405→      (s) => s.ts.getTime() >= fromMs && s.ts.getTime() < toMs
   406→    );
   407→    if (inWindow.length === 0) return null;
   408→    return inWindow.reduce((sum, s) => sum + s.value, 0) / inWindow.length;
   409→  };
   410→
   411→  const breaches = (rule: EvalRule, device: EvalDevice): { breached: boolean; observed: number | null } => {
   412→    if (rule.metric === "AVAILABILITY") {
   413→      const availability = availabilityOf(device, now);
   414→      return {
   415→        breached: compare(availability, rule.operator, rule.threshold),
   416→        observed: availability,
   417→      };
   418→    }
   419→    const avg = windowAvg(
   420→      device.id,
   421→      rule.metric,
   422→      now.getTime() - rule.durationMinutes * 60_000,
   423→      now.getTime() + 1
   424→    );
   425→    return {
   426→      breached: avg !== null && compare(avg, rule.operator, rule.threshold),
   427→      observed: avg,
   428→    };
   429→  };
   430→
   431→  const dedupKeyFor = (rule: EvalRule, deviceId: string): string =>
   432→    `${deviceId}:${rule.metric}:${rule.id}`;
   433→
   434→  const notify = async (row: {
   435→    kind: string;
   436→    title: string;
   437→    body: string;
   438→    link?: string;
   439→    severity?: string;
   440→  }): Promise<void> => {
   441→    if (summary.notificationsCreated >= MAX_NOTIFICATIONS_PER_RUN) {
   442→      summary.caps.notifications = true;
   443→      return;
   444→    }
   445→    await db.notification.create({ data: { userId: null, ...row } });
   446→    summary.notificationsCreated += 1;
   447→  };
   448→
   449→  /* ── pass 1: fires / dedup / suppression ───────────────────────────── */
   450→  for (const { rule, device } of pairs) {
   451→    const { breached, observed } = breaches(rule, device);
   452→    const key = dedupKeyFor(rule, device.id);
   453→    const existing =
   454→      existingByKey.get(key) ?? existingByDeviceRule.get(`${device.id}:${rule.id}`);
   455→
   456→    if (!breached) continue;
   457→
   458→    if (existing) {
   459→      // DEDUP — refresh the open alert instead of creating a duplicate.
   460→      const suppressedByWindow =
   461→        existing.status === "SUPPRESSED" &&
   462→        (existing.suppressReason ?? "").startsWith(MAINTENANCE_REASON_PREFIX);
   463→      const suppressedByRoot =
   464→        existing.status === "SUPPRESSED" &&
   465→        (existing.suppressReason ?? "").startsWith(ROOT_SUPPRESS_PREFIX);
   466→
   467→      if (existing.status === "ACTIVE" || existing.status === "ACKNOWLEDGED") {
   468→        const inMaintenance = maintenanceFor(device);
   469→        if (existing.status === "ACTIVE" && inMaintenance) {
   470→          // Existing ACTIVE alerts that fall inside a window are suppressed
   471→          // too (ACKNOWLEDGED keeps human ownership — documented choice).
   472→          await db.alert.update({
   473→            where: { id: existing.id },
   474→            data: {
   475→              status: "SUPPRESSED",
   476→              suppressReason: `${MAINTENANCE_REASON_PREFIX}${inMaintenance}`,
   477→              lastSeen: now,
   478→              count: { increment: 1 },
   479→              dedupKey: key,
   480→            },
   481→          });
   482→          summary.suppressed += 1;
   483→        } else {
   484→          await db.alert.update({
   485→            where: { id: existing.id },
   486→            data: { lastSeen: now, count: { increment: 1 }, dedupKey: key },
   487→          });
   488→          summary.deduped += 1;
   489→        }
   490→        if (rule.metric === "AVAILABILITY") openRootByDevice.set(device.id, existing);
   491→        continue;
   492→      }
   493→
   494→      // SUPPRESSED — maintenance expiry re-activates; root release handled
   495→      // by the children sweep below (root resolves first, next pass cleans).
   496→      if (suppressedByWindow && !maintenanceFor(device)) {
   497→        await db.alert.update({
   498→          where: { id: existing.id },
   499→          data: {
   500→            status: "ACTIVE",
   501→            suppressReason: null,
   502→            lastSeen: now,
   503→            count: { increment: 1 },
   504→            dedupKey: key,
   505→          },
   506→        });
   507→        summary.deduped += 1;
   508→      } else {
   509→        await db.alert.update({
   510→          where: { id: existing.id },
   511→          data: { lastSeen: now, count: { increment: 1 }, dedupKey: key },
   512→        });
   513→        summary.deduped += 1;
   514→      }
   515→      continue;
   516→    }
   517→
   518→    /* NEW FIRE */
   519→    if (summary.fired + summary.childrenSuppressed >= MAX_NEW_ALERTS_PER_RUN) {
   520→      summary.caps.newAlerts = true;
   521→      continue;
   522→    }
   523→
   524→    const isAvailability = rule.metric === "AVAILABILITY";
   525→    const maintenance = maintenanceFor(device);
   526→    const root = isAvailability ? null : openRootByDevice.get(device.id) ?? null;
   527→
   528→    let status = "ACTIVE";
   529→    let suppressReason: string | null = null;
   530→    let parentAlertId: string | null = null;
   531→
   532→    if (maintenance) {
   533→      status = "SUPPRESSED";
   534→      suppressReason = `${MAINTENANCE_REASON_PREFIX}${maintenance}`;
   535→      summary.suppressed += 1;
   536→    } else if (root) {
   537→      // Dependent alert on an unreachable device → suppressed CHILD (G5).
   538→      status = "SUPPRESSED";
   539→      suppressReason = `${ROOT_SUPPRESS_PREFIX}${root.id}`;
   540→      parentAlertId = root.id;
   541→      summary.childrenSuppressed += 1;
   542→    } else {
   543→      summary.fired += 1;
   544→    }
   545→
   546→    const label = METRIC_LABELS[rule.metric] ?? { label: rule.metric, unit: "" };
   547→    const message = isAvailability
   548→      ? `Device unreachable — availability ${fmt(observed ?? 0)} (${rule.operator} ${rule.threshold}) for ${rule.durationMinutes} min — rule "${rule.name}"`
   549→      : `${label.label} ${fmt(observed ?? 0)}${label.unit} (${rule.operator} ${rule.threshold}${label.unit} for ${rule.durationMinutes} min) — rule "${rule.name}"`;
   550→
   551→    const created = await db.alert.create({
   552→      data: {
   553→        deviceId: device.id,
   554→        ruleId: rule.id,
   555→        severity: rule.severity,
   556→        message,
   557→        status,
   558→        suppressReason,
   559→        parentAlertId,
   560→        dedupKey: key,
   561→        firstSeen: now,
   562→        lastSeen: now,
   563→        count: 1,
   564→      },
   565→      select: { id: true },
   566→    });
   567→    if (isAvailability && status === "ACTIVE") {
   568→      // Root for this run — children fired later in the same pass (or the
   569→      // children sweep below) parent onto this real row id.
   570→      openRootByDevice.set(device.id, {
   571→        id: created.id,
   572→        deviceId: device.id,
   573→        ruleId: rule.id,
   574→        dedupKey: key,
   575→        status: "ACTIVE",
   576→        suppressReason: null,
   577→        parentAlertId: null,
   578→      });
   579→    }
   580→
   581→    await db.auditEvent.create({
   582→      data: {
   583→        actorName: "system:alert-engine",
   584→        action: status === "ACTIVE" ? "ALERT_FIRED" : "ALERT_SUPPRESSED",
   585→        resourceType: "Alert",
   586→        resourceId: created.id,
   587→        resourceLabel: device.hostname,
   588→        result: "SUCCESS",
   589→        correlationId,
   590→        afterJson: JSON.stringify({
   591→          rule: rule.name,
   592→          metric: rule.metric,
   593→          severity: rule.severity,
   594→          status,
   595→          observed,
   596→          suppressReason,
   597→          parentAlertId,
   598→        }),
   599→      },
   600→    });
   601→
   602→    /* Incident auto-creation + notifications — only for real fires. */
   603→    if (status === "ACTIVE" && (rule.severity === "CRITICAL" || rule.severity === "HIGH")) {
   604→      const openIncident = await db.incidentDevice.count({
   605→        where: {
   606→          deviceId: device.id,
   607→          incident: { status: { notIn: ["RESOLVED", "CLOSED"] } },
   608→        },
   609→      });
   610→      if (
   611→        openIncident === 0 &&
   612→        summary.incidentsCreated < MAX_INCIDENTS_PER_RUN
   613→      ) {
   614→        const result = await createIncidentForAlert({
   615→          alert: {
   616→            id: created.id,
   617→            severity: rule.severity,
   618→            message,
   619→            deviceId: device.id,
   620→          },
   621→          device: {
   622→            id: device.id,
   623→            hostname: device.hostname,
   624→            siteId: device.siteId,
   625→          },
   626→          source: "ALERT",
   627→          correlationId,
   628→        });
   629→        if (result.created) {
   630→          summary.incidentsCreated += 1;
   631→          await notify({
   632→            kind: "INCIDENT",
   633→            title: `Incident ${result.incident?.number} auto-created`,
   634→            body: `${device.hostname}: ${message}`,
   635→            link: "ops.incidents",
   636→            severity: result.incident?.severity,
   637→          });
   638→        }
   639→      }
   640→      await notify({
   641→        kind: "ALERT",
   642→        title: `${rule.severity} alert: ${label.label} on ${device.hostname}`,
   643→        body: message,
   644→        link: "ops.alerts",
   645→        severity: rule.severity,
   646→      });
   647→    }
   648→  }
   649→
   650→  /* ── children sweep: existing ACTIVE alerts on unreachable devices ── */
   651→  for (const [deviceId, root] of openRootByDevice) {
   652→    if (!root.id) continue;
   653→    const device = deviceById.get(deviceId);
   654→    if (!device || maintenanceFor(device)) continue;
   655→    const activeOnDevice = existingRows.filter(
   656→      (row) =>
   657→        row.deviceId === deviceId &&
   658→        row.id !== root.id &&
   659→        (row.status === "ACTIVE" || row.status === "ACKNOWLEDGED") &&
   660→        !row.dedupKey?.includes(":AVAILABILITY:")
   661→    );
   662→    for (const row of activeOnDevice) {
   663→      if (row.status !== "ACTIVE") continue; // ACKNOWLEDGED keeps ownership
   664→      await db.alert.update({
   665→        where: { id: row.id },
   666→        data: {
   667→          status: "SUPPRESSED",
   668→          suppressReason: `${ROOT_SUPPRESS_PREFIX}${root.id}`,
   669→          parentAlertId: root.id,
   670→        },
   671→      });
   672→      summary.suppressed += 1;
   673→    }
   674→  }
   675→
   676→  /* ── pass 2: auto-resolve ──────────────────────────────────────────── */
   677→  const ruleById = new Map(rules.map((r) => [r.id as string, r]));
   678→  for (const existing of existingRows) {
   679→    if (existing.status !== "ACTIVE" && existing.status !== "ACKNOWLEDGED") continue;
   680→    if (!existing.ruleId) continue;
   681→    const rule = ruleById.get(existing.ruleId);
   682→    if (!rule) continue; // rule deleted/inactive — alert is human-managed
   683→    const device = deviceById.get(existing.deviceId);
   684→    if (!device) continue;
   685→
   686→    let recovered = false;
   687→    if (rule.metric === "AVAILABILITY") {
   688→      recovered = availabilityOf(device, now) === 1;
   689→    } else {
   690→      // 2 consecutive windows OK: recent window AND the one before it.
   691→      const recent = windowAvg(
   692→        device.id,
   693→        rule.metric,
   694→        now.getTime() - rule.durationMinutes * 60_000,
   695→        now.getTime() + 1
   696→      );
   697→      const previous = windowAvg(
   698→        device.id,
   699→        rule.metric,
   700→        now.getTime() - 2 * rule.durationMinutes * 60_000,
   701→        now.getTime() - rule.durationMinutes * 60_000
   702→      );
   703→      if (recent === null || previous === null) continue; // sparse data → skip
   704→      recovered =
   705→        !compare(recent, rule.operator, rule.threshold) &&
   706→        !compare(previous, rule.operator, rule.threshold);
   707→    }
   708→    if (!recovered) continue;
   709→
   710→    await db.alert.update({
   711→      where: { id: existing.id },
   712→      data: { status: "RESOLVED", suppressReason: null },
   713→    });
   714→    summary.resolved += 1;
   715→    await db.auditEvent.create({
   716→      data: {
   717→        actorName: "system:alert-engine",
   718→        action: "ALERT_RESOLVED",
   719→        resourceType: "Alert",
   720→        resourceId: existing.id,
   721→        resourceLabel: device.hostname,
   722→        result: "SUCCESS",
   723→        correlationId,
   724→        afterJson: JSON.stringify({
   725→          rule: rule.name,
   726→          metric: rule.metric,
   727→          reason: "condition recovered",
   728→        }),
   729→      },
   730→    });
   731→  }
   732→
   733→  /* ── summary audit ─────────────────────────────────────────────────── */
   734→  await db.auditEvent.create({
   735→    data: {
   736→      actorName: "system:alert-engine",
   737→      action: "ALERT_EVALUATION_COMPLETED",
   738→      resourceType: "AlertRule",
   739→      result: "SUCCESS",
   740→      correlationId,
   741→      afterJson: JSON.stringify({
   742→        triggeredBy,
   743→        rulesEvaluated: summary.rulesEvaluated,
   744→        devicesConsidered: summary.devicesConsidered,
   745→        fired: summary.fired,
   746→        deduped: summary.deduped,
   747→        suppressed: summary.suppressed,
   748→        childrenSuppressed: summary.childrenSuppressed,
   749→        resolved: summary.resolved,
   750→        incidentsCreated: summary.incidentsCreated,
   751→        notificationsCreated: summary.notificationsCreated,
   752→      }),
   753→    },
   754→  });
   755→
   756→  return summary;
   757→}
