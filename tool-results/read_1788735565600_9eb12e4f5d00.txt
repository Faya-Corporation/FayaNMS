   109→function formatDate(value: string | null | undefined, pattern = "EEE, MMM d, HH:mm"): string {
   110→  if (!value) return "—";
   111→  const date = new Date(value);
   112→  return Number.isNaN(date.getTime()) ? "—" : format(date, pattern);
   113→}
   114→
   115→/** Human step duration between startedAt/finishedAt. */
   116→function stepDuration(step: ChangeDetailStep): string | null {
   117→  if (!step.startedAt || !step.finishedAt) return null;
   118→  const start = new Date(step.startedAt).getTime();
   119→  const end = new Date(step.finishedAt).getTime();
   120→  if (Number.isNaN(start) || Number.isNaN(end) || end < start) return null;
   121→  const seconds = Math.max(1, Math.round((end - start) / 1000));
   122→  return seconds < 60 ? `${seconds}s` : `${Math.floor(seconds / 60)}m ${seconds % 60}s`;
   123→}

   170→/** Vertical step timeline — live-updates while the engine drives the change. */
   171→function StepTimeline({ steps }: { steps: ChangeDetailStep[] }) {
   172→  return (
   173→    <ol className="relative flex flex-col gap-0 ps-1">
   174→      {steps.map((step, index) => {
   175→        const typeConfig = lookupStatusConfig(CHANGE_STEP_TYPE_UI, step.type);
   176→        const statusConfig = lookupStatusConfig(CHANGE_STEP_STATUS_UI, step.status);
   177→        const running = step.status === "RUNNING";
   178→        const last = index === steps.length - 1;
   179→        const duration = stepDuration(step);
   180→        return (
   181→          <li className="relative flex gap-3 pb-4" key={step.id}>
   182→            {!last && (
   183→              <span
   184→                aria-hidden="true"
   185→                className="absolute start-[15px] top-8 h-[calc(100%-16px)] w-px bg-border"
   186→              />
   187→            )}
   188→            <span className="relative z-10 flex size-8 shrink-0 items-center justify-center rounded-full border bg-card">
   189→              {running ? (
   190→                <LoaderCircle
   191→                  aria-hidden="true"
   192→                  className="size-4 animate-spin text-info"
   193→                />
   194→              ) : (
   195→                <StatusIcon className={typeConfig.iconClass} icon={typeConfig.icon} />
   196→              )}
   197→            </span>
   198→            <div className="min-w-0 flex-1 pt-1">
   199→              <div className="flex flex-wrap items-center gap-2">
   200→                <span className="font-tech text-xs tabular-nums text-muted-foreground">
   201→                  {step.order}.
   202→                </span>
   203→                <span className="text-sm font-medium">{step.name}</span>
   204→                <Badge className="font-tech" variant="outline">
   205→                  {step.type}
   206→                </Badge>
   207→                <StatusBadge config={statusConfig} />
   208→                {running && (
   209→                  <span className="inline-flex items-center gap-1.5 text-xs font-medium text-info">
   210→                    <span
   211→                      aria-hidden="true"
   212→                      className="size-1.5 animate-pulse rounded-full bg-info"
   213→                    />
   214→                    running…
   215→                  </span>
   216→                )}
   217→              </div>
   218→              {(step.startedAt || step.finishedAt) && (
   219→                <p className="mt-1 text-xs tabular-nums text-muted-foreground">
   220→                  {step.startedAt && `started ${formatDate(step.startedAt, "MMM d HH:mm")}`}
   221→                  {step.startedAt && step.finishedAt && " · "}
   222→                  {step.finishedAt && `finished ${formatDate(step.finishedAt, "MMM d HH:mm")}`}
   223→                  {duration && ` · took ${duration}`}
   224→                </p>
   225→              )}
   226→              {step.error && <StepOutputBlock text={step.error} tone="error" />}
   227→              {step.output && <StepOutputBlock text={step.output} tone="output" />}
   228→            </div>
   229→          </li>
   230→        );
   231→      })}
   232→    </ol>
   233→  );
   234→}

   236→/**

   239→ * { changeId }. 4-b adds the approval actions (SoD-aware), the guarded

   294→      <div className="flex flex-col gap-4" aria-busy="true" aria-live="polite">

   302→      </div>
   303→    );
   304→  }
   305→
   306→  const change = detail.data;
   307→  const canEdit = EDITABLE_STATUSES.includes(change.status);
   308→  const canCancel = CANCELLABLE_STATUSES.includes(change.status);
   309→  const canExecute =
   310→    EXECUTABLE_STATUSES.includes(change.status) &&
   311→    change.approvals.every(
   312→      (approval) => approval.status === "APPROVED" || approval.status === "NOT_REQUIRED"
   313→    ) &&
   314→    change.approvals.length > 0;
   315→  const canClose = change.status === "SUCCESSFUL";
   316→  const executing = EXECUTION_STATUSES.includes(change.status);
   317→  const stepsDone = change.steps.filter(
   318→    (step) => step.status === "PASSED" || step.status === "SKIPPED"
   319→  ).length;
   320→  const pendingApprovals = change.approvals.filter(
   321→    (approval) => approval.status === "PENDING"
   322→  ).length;
   323→  const requiredLevels = approvalLevelsFor(change.riskLevel);
   324→  const missingApprovalRows = requiredLevels.filter(
   325→    (level) => !change.approvals.some((approval) => approval.level === level)
   326→  );
   327→  const showFailedBanner =
   328→    FAILED_STATUSES.includes(change.status) && change.incidents.length === 0;

   330→  /** SoD pre-check — the requester cannot decide a HIGH/CRITICAL change. */
   331→  const sodBlocked = Boolean(
   332→    actingUser &&
   333→      change.requester.id === actingUser.id &&
   334→      SOD_GATED_RISK_LEVELS.includes(change.riskLevel)

   340→      { id: change.id, data: { action: confirmAction } },

   342→        onSuccess: () => {

   356→      {
   357→        changeId: change.id,
   358→        payload: {
   359→          level: decision.level,
   360→          decision: decision.decision,
   361→          comment: trimmed || undefined,
   362→          actAsUserId: actAsUserId || undefined,
   363→        },
   364→      },
   365→      {
   366→        onSuccess: () => {
   367→          setDecision(null);
   368→          setComment("");

   370→        },
   371→      }
   372→    );
   373→  };

   393→          <Button
   394→            onClick={() => setActiveView("changes.all")}
   395→            size="sm"
   396→            variant="ghost"
   397→          >
   398→            <ArrowLeft aria-hidden="true" />
   399→            All Changes
   400→          </Button>
   401→        </div>
   402→        <div className="flex flex-col gap-3 md:flex-row md:items-end md:justify-between">
   403→          <div className="min-w-0 space-y-1.5">
   404→            <div className="flex flex-wrap items-center gap-2">
   405→              <span className="font-tech text-sm font-semibold ltr-technical text-muted-foreground">
   406→                {change.number}
   407→              </span>
   408→              <StatusBadge
   409→                config={lookupStatusConfig(CHANGE_STATUS_UI, change.status)}
   410→              />
   411→              <StatusBadge
   412→                config={lookupStatusConfig(CHANGE_TYPE_UI, change.type)}
   413→              />
   414→              <ChangeRiskBadge value={change.riskLevel} />

   427→              <>
   428→                <Button
   429→                  onClick={() => setWizardOpen(true)}
   430→                  size="sm"
   431→                  variant="outline"
   432→                >
   433→                  <Pencil aria-hidden="true" />

   448→                <Play aria-hidden="true" />

   455→                size="sm"
