   105→function csvEscape(value: string | number | null | undefined): string {
   106→  if (value === null || value === undefined) return "";
   107→  const text = String(value);
   108→  return /[",\n]/.test(text) ? `"${text.replace(/"/g, '""')}"` : text;
   109→}

   111→function buildDeviceCsv(rows: DeviceRow[]): string {
   112→  const header = [
   113→    "hostname",
   114→    "display_name",
   115→    "mgmt_ip",
   116→    "vendor",
   117→    "model",
   118→    "site",
   119→    "status",
   120→    "criticality",
   121→    "health_score",
   122→    "backup_compliance",
   123→    "last_backup_at",
   124→    "last_seen",
   125→    "interfaces",
   126→    "snapshots",
   127→    "alerts",
   128→  ];
   129→  const lines = rows.map((device) =>
   130→    [
   131→      device.hostname,
   132→      device.displayName,
   133→      device.mgmtIp,
   134→      device.vendor?.name ?? "",
   135→      device.model,
   136→      device.site?.name ?? "",
   137→      device.status,
   138→      device.criticality,
   139→      device.healthScore,
   140→      device.backupCompliance,
   141→      device.lastBackupAt,
   142→      device.lastSeen,
   143→      device._count.interfaces,
   144→      device._count.snapshots,
   145→      device._count.alerts,
   146→    ]
   147→      .map(csvEscape)
   148→      .join(",")
   149→  );
   150→  return [header.join(","), ...lines].join("\n");
   151→}

   153→function downloadCsv(filename: string, csv: string) {
   154→  const blob = new Blob([csv], { type: "text/csv;charset=utf-8;" });
   155→  const url = URL.createObjectURL(blob);
   156→  const anchor = document.createElement("a");
   157→  anchor.href = url;
   158→  anchor.download = filename;
   159→  document.body.appendChild(anchor);
   160→  anchor.click();
   161→  anchor.remove();
   162→  URL.revokeObjectURL(url);
   163→}

   169→function SortableHead({
   170→  label,
   171→  field,
   172→  sort,
   173→  dir,
   174→  onSort,
   175→  className,
   176→}: {
   177→  label: string;
   178→  field: DeviceSortField;
   179→  sort: DeviceSortField;
   180→  dir: "asc" | "desc";
   181→  onSort: (field: DeviceSortField) => void;
   182→  className?: string;
   183→}) {
   184→  const active = sort === field;
   185→  const Icon = !active ? ChevronsUpDown : dir === "asc" ? ArrowUp : ArrowDown;
   186→  return (
   187→    <TableHead aria-sort={active ? (dir === "asc" ? "ascending" : "descending") : "none"} className={className}>
   188→      <button
   189→        aria-label={`Sort by ${label}`}
   190→        className={cn(
   191→          "inline-flex items-center gap-1 rounded-sm text-xs font-medium transition-colors hover:text-foreground",
   192→          active ? "text-foreground" : "text-muted-foreground"
   193→        )}
   194→        onClick={() => onSort(field)}
   195→        type="button"
   196→      >
   197→        {label}
   198→        <Icon aria-hidden="true" className="size-3" />
   199→      </button>
   200→    </TableHead>
   201→  );
   202→}
