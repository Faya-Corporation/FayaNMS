    64→/**
    65→ * Discovery (Phase 2-c): queue simulated subnet scans on the worker, watch
    66→ * the scan history progress live, and import selected candidates into the
    67→ * device inventory. Candidates are persistence-free — they live in the
    68→ * DISCOVERY job's resultJson until imported.
    69→ */
    70→
    71→const IPV4_OCTET = "(25[0-5]|2[0-4]\\d|1\\d\\d|[1-9]?\\d)";
    72→const CIDR_PATTERN = new RegExp(
    73→  `^${IPV4_OCTET}\\.${IPV4_OCTET}\\.${IPV4_OCTET}\\.${IPV4_OCTET}\\/(3[0-2]|[12]?\\d)$`
    74→);
    75→
    76→const scanFormSchema = z.object({
    77→  subnets: z
    78→    .array(z.string().trim().regex(CIDR_PATTERN, "Not a valid CIDR (a.b.c.d/prefix)"))
    79→    .min(1, "At least one subnet is required")
    80→    .max(8, "A scan is limited to 8 subnets"),
    81→  name: z.string().trim().max(120).optional(),
    82→});
    83→
    84→/** "Imported" chip — neutral token family per the status.ts token classes. */
    85→const IMPORTED_CHIP: StatusBadgeConfig = {
    86→  key: "IMPORTED",
    87→  label: "Imported",
    88→  token: "neutral",
    89→  icon: "Check",
    90→  dotClass: "bg-neutral",
    91→  badgeClass: "bg-neutral-subtle text-neutral border-neutral/25",
    92→  iconClass: "text-neutral",
    93→};
    94→
    95→/** "New" chip for not-yet-imported candidates. */
    96→const NEW_CHIP: StatusBadgeConfig = {
    97→  key: "NEW",
    98→  label: "New",
    99→  token: "info",
   100→  icon: "Sparkles",
   101→  dotClass: "bg-info",
   102→  badgeClass: "bg-info-subtle text-info border-info/25",
   103→  iconClass: "text-info",
   104→};
   105→
   106→const CRITICALITIES: { value: "LOW" | "MEDIUM" | "HIGH" | "CRITICAL"; label: string }[] = [
   107→  { value: "LOW", label: "Low" },
   108→  { value: "MEDIUM", label: "Medium" },
   109→  { value: "HIGH", label: "High" },
   110→  { value: "CRITICAL", label: "Critical" },
   111→];
   112→
   113→function durationLabel(ms: number | null): string {
   114→  if (ms === null || ms < 0) return "—";
   115→  return ms < 1000 ? `${ms} ms` : `${(ms / 1000).toFixed(1)} s`;
   116→}
   117→
   118→export function DiscoveryView() {
   119→  const { toast } = useToast();
   120→  const jobs = useDiscoveryJobs();
   121→  const meta = useMeta();
   122→  const startScan = useStartScan();
   123→  const importCandidates = useImportCandidates();
   124→
   125→  const [scanOpen, setScanOpen] = useState(false);
   126→  const [scanName, setScanName] = useState("");
   127→  const [subnetsText, setSubnetsText] = useState("");
   128→  const [selectedJobId, setSelectedJobId] = useState<string | null>(null);
   129→  const [selectedCandidates, setSelectedCandidates] = useState<Set<string>>(new Set());
   130→  const [importOpen, setImportOpen] = useState(false);
   131→  const [importSiteId, setImportSiteId] = useState("");
   132→  const [importCredentialId, setImportCredentialId] = useState("");
   133→  const [importCriticality, setImportCriticality] =
   134→    useState<"LOW" | "MEDIUM" | "HIGH" | "CRITICAL">("MEDIUM");
   135→  const [importManaged, setImportManaged] = useState(true);
   136→
   137→  const rows = jobs.data ?? [];
   138→
   139→  // Selected scan: the explicitly chosen job, else the newest scan that
   140→  // already carries candidates (derived — no selection side-effects).
   141→  const selectedJob = useMemo(() => {
   142→    if (selectedJobId) {
   143→      const explicit = rows.find((job) => job.id === selectedJobId);
   144→      if (explicit) return explicit;
   145→    }
   146→    return (
   147→      rows.find(
   148→        (job) => job.status === "SUCCEEDED" && job.candidateCount > 0
   149→      ) ?? null
   150→    );
   151→  }, [rows, selectedJobId]);
   152→
   153→  /** History row click — switches the scan and clears candidate selection. */
   154→  const handleSelectJob = (jobId: string) => {
   155→    setSelectedJobId(jobId);
   156→    setSelectedCandidates(new Set());
   157→  };
   158→
   159→  const importable = useMemo(
   160→    () =>
   161→      (selectedJob?.candidates ?? []).filter(
   162→        (candidate) => !candidate.imported
   163→      ),
   164→    [selectedJob]
   165→  );
   166→
   167→  const pendingScan = startScan.isPending;
   168→  const pendingImport = importCandidates.isPending;
   169→
   170→  // New-scan dialog: parse the textarea one CIDR per line and validate.
   171→  const scanIssues = useMemo(() => {
   172→    const lines = subnetsText
   173→      .split(/\r?\n/)
   174→      .map((line) => line.trim())
   175→      .filter((line) => line.length > 0);
   176→    return lines.filter((line) => !CIDR_PATTERN.test(line));
   177→  }, [subnetsText]);
   178→
   179→  const handleStartScan = () => {
   180→    const lines = subnetsText
   181→      .split(/\r?\n/)
   182→      .map((line) => line.trim())
   183→      .filter((line) => line.length > 0);
   184→    const parsed = scanFormSchema.safeParse({
   185→      subnets: lines,
   186→      name: scanName.trim() || undefined,
   187→    });
   188→    if (!parsed.success) {
   189→      toast({
   190→        title: "Check the scan settings",
   191→        description: parsed.error.issues[0]?.message ?? "Invalid scan",
   192→        variant: "destructive",
   193→      });
   194→      return;
   195→    }
   196→    startScan.mutate(parsed.data, {
   197→      onSuccess: (result) => {
   198→        setScanOpen(false);
   199→        setScanName("");
   200→        setSubnetsText("");
   201→        setSelectedCandidates(new Set());
   202→        setSelectedJobId(result.jobId);
   203→      },
   204→    });
   205→  };
   206→
   207→  const handleToggleCandidate = (ip: string, checked: boolean) => {
   208→    setSelectedCandidates((current) => {
   209→      const next = new Set(current);
   210→      if (checked) {
   211→        next.add(ip);
   212→      } else {
   213→        next.delete(ip);
   214→      }
   215→      return next;
   216→    });
   217→  };
   218→
   219→  const handleImport = () => {
   220→    if (!selectedJob) return;
   221→    importCandidates.mutate(
   222→      {
   223→        jobId: selectedJob.id,
   224→        ips: Array.from(selectedCandidates),
   225→        siteId: importSiteId || undefined,
   226→        credentialProfileId: importCredentialId || undefined,
   227→        criticality: importCriticality,
   228→        managed: importManaged,
   229→      },
   230→      {
   231→        onSuccess: (result) => {
   232→          setImportOpen(false);
   233→          setSelectedCandidates(new Set());
   234→          if (result.skipped.length > 0) {
   235→            toast({
   236→              title: `${result.skipped.length} candidate${result.skipped.length === 1 ? "" : "s"} skipped`,
   237→              description: result.skipped
   238→                .slice(0, 3)
   239→                .map((entry) => `${entry.ip}: ${entry.reason}`)
   240→                .join(" · "),
   241→            });
   242→          }
   243→        },
   244→      }
   245→    );
   246→  };
   247→
   248→  return (
   249→    <div className="flex flex-col gap-4">
   250→      <PageHeader
   251→        description="Scan subnets for candidate devices and import them into the inventory"
   252→        primaryAction={
   253→          <Button onClick={() => setScanOpen(true)}>
   254→            <Radar aria-hidden="true" />
   255→            New Scan
   256→          </Button>
   257→        }
   258→        title="Discovery"
   259→      />
   260→
   261→      {/* Scan history */}
   262→      <SectionCard
   263→        contentClassName="p-0"
   264→        description="The worker sweeps each subnet and reports candidates — polling is live while a scan runs"
   265→        title="Scan history"
   266→      >
   267→        {jobs.isError ? (
   268→          <div className="p-4">
   269→            <ErrorState
   270→              onRetry={() => void jobs.refetch()}
   271→              reason={jobs.error.message}
   272→              title="Scan history could not be loaded"
   273→            />
   274→          </div>
   275→        ) : jobs.isLoading ? (
   276→          <div className="flex flex-col gap-2 p-4">
   277→            {Array.from({ length: 4 }).map((_, index) => (
   278→              <div key={index} className="h-12 animate-pulse rounded-md bg-muted/60" />
   279→            ))}
   280→          </div>
   281→        ) : rows.length === 0 ? (
   282→          <div className="p-4">
   283→            <EmptyState
   284→              description="Queue a scan against one or more subnets (one CIDR per line, e.g. 10.60.0.0/24) — discovered candidates appear here."
   285→              icon={Radar}
   286→              title="No discovery scans yet"
   287→            />
   288→          </div>
   289→        ) : (
   290→          <div className="overflow-x-auto">
   291→            <Table className="min-w-[860px]">
   292→              <TableHeader>
   293→                <TableRow className="hover:bg-transparent">
   294→                  <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Scan</TableHead>
   295→                  <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Status</TableHead>
   296→                  <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Subnets</TableHead>
   297→                  <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Candidates</TableHead>
   298→                  <TableHead className="hidden h-(--density-row-h) px-(--density-cell-x) md:table-cell">
   299→                    Duration
   300→                  </TableHead>
   301→                  <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Created</TableHead>
   302→                </TableRow>
   303→              </TableHeader>
   304→              <TableBody>
   305→                {rows.map((job) => {
   306→                  const running = job.status === "RUNNING" || job.status === "QUEUED";
   307→                  return (
   308→                    <TableRow
   309→                      className={cn("cursor-pointer", job.id === selectedJob?.id && "bg-primary/5")}
   310→                      data-state={job.id === selectedJob?.id ? "selected" : undefined}
   311→                      key={job.id}
   312→                      onClick={() => handleSelectJob(job.id)}
   313→                    >
   314→                      <TableCell className="h-(--density-row-h) px-(--density-cell-x)">
   315→                        <div className="flex flex-col">
   316→                          <span className="text-sm font-medium">
   317→                            {job.name ?? "Ad-hoc scan"}
   318→                          </span>
   319→                          <span className="font-tech text-xs ltr-technical text-muted-foreground">
   320→                            {job.correlationId}
   321→                          </span>
   322→                        </div>
   323→                      </TableCell>
   324→                      <TableCell className="h-(--density-row-h) px-(--density-cell-x)">
   325→                        <div className="flex flex-col gap-1.5">
   326→                          <JobStatusBadge value={job.status} />
   327→                          {running && (
   328→                            <Progress
   329→                              aria-label={`Scan ${job.correlationId} progress`}
   330→                              className="h-1.5 w-28"
   331→                              value={job.progress}
   332→                            />
   333→                          )}
   334→                        </div>
   335→                      </TableCell>
   336→                      <TableCell className="h-(--density-row-h) px-(--density-cell-x)">
   337→                        <div className="flex flex-wrap gap-1">
   338→                          {job.subnets.map((subnet) => (
   339→                            <Badge
   340→                              className="font-tech ltr-technical"
   341→                              key={subnet}
   342→                              variant="outline"
   343→                            >
   344→                              {subnet}
   345→                            </Badge>
   346→                          ))}
   347→                        </div>
   348→                      </TableCell>
   349→                      <TableCell className="h-(--density-row-h) px-(--density-cell-x) tabular-nums">
   350→                        {job.status === "SUCCEEDED" ? (
   351→                          <span>
   352→                            {job.candidateCount}
   353→                            {job.importedCount > 0 && (
   354→                              <span className="text-muted-foreground">
   355→                                {" "}
   356→                                ({job.importedCount} imported)
   357→                              </span>
   358→                            )}
   359→                          </span>
   360→                        ) : (
   361→                          "—"
   362→                        )}
   363→                      </TableCell>
   364→                      <TableCell className="hidden h-(--density-row-h) px-(--density-cell-x) tabular-nums md:table-cell">
   365→                        {durationLabel(job.durationMs)}
   366→                      </TableCell>
   367→                      <TableCell className="h-(--density-row-h) px-(--density-cell-x) text-xs text-muted-foreground">
   368→                        {formatDistanceToNow(new Date(job.createdAt), { addSuffix: true })}
   369→                      </TableCell>
   370→                    </TableRow>
   371→                  );
   372→                })}
   373→              </TableBody>
   374→            </Table>
   375→          </div>
   376→        )}
   377→      </SectionCard>
   378→
   379→      {/* Candidates of the selected scan */}
   380→      {selectedJob && (
   381→        <SectionCard
   382→          contentClassName="p-0"
   383→          actions={
   384→            <span className="font-tech text-xs ltr-technical text-muted-foreground">
   385→              {selectedJob.correlationId}
   386→            </span>
   387→          }
   388→          description={
   389→            selectedJob.status === "SUCCEEDED"
   390→              ? `${selectedJob.candidateCount} candidate${selectedJob.candidateCount === 1 ? "" : "s"} · ${selectedJob.subnets.join(", ")}`
   391→              : "Candidates appear when the scan finishes"
   392→          }
   393→          title="Candidates"
   394→        >
   395→          {selectedJob.status !== "SUCCEEDED" ? (
   396→            <div className="p-4">
   397→              <EmptyState
   398→                description={
   399→                  selectedJob.error
   400→                    ? `Last error: ${selectedJob.error}`
   401→                    : "The scan is queued or running — the table fills in as soon as the worker reports."
   402→                }
   403→                icon={LoaderCircle}
   404→                title="Scan not finished yet"
   405→              />
   406→            </div>
   407→          ) : selectedJob.candidateCount === 0 ? (
   408→            <div className="p-4">
   409→              <EmptyState
   410→                description="The scan finished without discovering candidates. Try a different subnet range."
   411→                icon={SearchX}
   412→                title="No candidates in this scan"
   413→              />
   414→            </div>
   415→          ) : (
   416→            <>
   417→              {/* Bulk import bar */}
   418→              {selectedCandidates.size > 0 && (
   419→                <div
   420→                  aria-live="polite"
   421→                  className="flex flex-wrap items-center gap-2 border-b border-t bg-primary/5 px-4 py-2.5"
   422→                >
   423→                  <span className="text-sm font-medium">
   424→                    {selectedCandidates.size} candidate
   425→                    {selectedCandidates.size === 1 ? "" : "s"} selected
   426→                  </span>
   427→                  <Button disabled={pendingImport} onClick={() => setImportOpen(true)} size="sm">
   428→                    <Crosshair aria-hidden="true" />
   429→                    Import {selectedCandidates.size} candidate
   430→                    {selectedCandidates.size === 1 ? "" : "s"}
   431→                  </Button>
   432→                  <Button
   433→                    onClick={() => setSelectedCandidates(new Set())}
   434→                    size="sm"
   435→                    variant="ghost"
   436→                  >
   437→                    Clear
   438→                  </Button>
   439→                </div>
   440→              )}
   441→
   442→              <div className="overflow-x-auto">
   443→                <Table className="min-w-[980px]">
   444→                  <TableHeader>
   445→                    <TableRow className="hover:bg-transparent">
   446→                      <TableHead className="h-(--density-row-h) w-10 px-(--density-cell-x)">
   447→                        <Checkbox
   448→                          aria-label="Select all importable candidates"
   449→                          checked={
   450→                            importable.length > 0 &&
   451→                            importable.every((candidate) =>
   452→                              selectedCandidates.has(candidate.ip)
   453→                            )
   454→                              ? true
   455→                              : selectedCandidates.size > 0
   456→                                ? "indeterminate"
   457→                                : false
   458→                          }
   459→                          disabled={importable.length === 0}
   460→                          onCheckedChange={(checked) => {
   461→                            setSelectedCandidates(
   462→                              checked === true
   463→                                ? new Set(importable.map((candidate) => candidate.ip))
   464→                                : new Set()
   465→                            );
   466→                          }}
   467→                        />
   468→                      </TableHead>
   469→                      <TableHead className="h-(--density-row-h) px-(--density-cell-x)">IP</TableHead>
   470→                      <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Hostname</TableHead>
   471→                      <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Vendor</TableHead>
   472→                      <TableHead className="hidden h-(--density-row-h) px-(--density-cell-x) md:table-cell">Model</TableHead>
   473→                      <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Confidence</TableHead>
   474→                      <TableHead className="hidden h-(--density-row-h) px-(--density-cell-x) lg:table-cell">Ports / protocols</TableHead>
   475→                      <TableHead className="hidden h-(--density-row-h) px-(--density-cell-x) xl:table-cell">OS fingerprint</TableHead>
   476→                      <TableHead className="h-(--density-row-h) px-(--density-cell-x)">Imported</TableHead>
   477→                    </TableRow>
   478→                  </TableHeader>
   479→                  <TableBody>
   480→                    {selectedJob.candidates.map((candidate) => (
   481→                      <CandidateRow
   482→                        candidate={candidate}
   483→                        key={candidate.ip}
   484→                        onToggle={handleToggleCandidate}
   485→                        selected={selectedCandidates.has(candidate.ip)}
   486→                      />
   487→                    ))}
   488→                  </TableBody>
   489→                </Table>
   490→              </div>
   491→            </>
   492→          )}
   493→        </SectionCard>
   494→      )}
   495→
   496→      {/* New scan dialog */}
   497→      <Dialog onOpenChange={setScanOpen} open={scanOpen}>
   498→        <DialogContent className="sm:max-w-md">
   499→          <DialogHeader>
   500→            <DialogTitle>New discovery scan</DialogTitle>
   501→            <DialogDescription>
   502→              One subnet per line in CIDR notation. The simulation worker scans
   503→              each subnet and reports candidate devices.
   504→            </DialogDescription>
   505→          </DialogHeader>
   506→          <div className="flex flex-col gap-4">
   507→            <div className="flex flex-col gap-1.5">
   508→              <Label htmlFor="scan-name">Name (optional)</Label>
   509→              <Input
   510→                id="scan-name"
   511→                onChange={(event) => setScanName(event.target.value)}
   512→                placeholder="Branch sweep — September"
   513→                value={scanName}
   514→              />
   515→            </div>
   516→            <div className="flex flex-col gap-1.5">
   517→              <Label htmlFor="scan-subnets">Subnets *</Label>
   518→              <Textarea
   519→                aria-invalid={scanIssues.length > 0}
   520→                className="font-tech ltr-technical"
   521→                id="scan-subnets"
   522→                onChange={(event) => setSubnetsText(event.target.value)}
   523→                placeholder={"10.60.0.0/24\n10.70.0.0/24"}
   524→                rows={4}
   525→                value={subnetsText}
   526→              />
   527→              {scanIssues.length > 0 ? (
   528→                <p className="text-xs text-danger">
   529→                  Invalid subnets: {scanIssues.join(", ")} — expected a.b.c.d/prefix.
   530→                </p>
   531→              ) : (
   532→                <p className="text-xs text-muted-foreground">
   533→                  Up to 8 subnets per scan (e.g. 10.60.0.0/24).
   534→                </p>
   535→              )}
   536→            </div>
   537→          </div>
   538→          <DialogFooter>
   539→            <Button onClick={() => setScanOpen(false)} type="button" variant="outline">
   540→              Cancel
   541→            </Button>
   542→            <Button
   543→              disabled={pendingScan || scanIssues.length > 0}
   544→              onClick={handleStartScan}
   545→              type="button"
   546→            >
   547→              {pendingScan && <LoaderCircle aria-hidden="true" className="animate-spin" />}
   548→              Queue scan
   549→            </Button>
   550→          </DialogFooter>
   551→        </DialogContent>
   552→      </Dialog>
   553→
   554→      {/* Import dialog */}
   555→      <Dialog onOpenChange={setImportOpen} open={importOpen}>
   556→        <DialogContent className="sm:max-w-md">
   557→          <DialogHeader>
   558→            <DialogTitle>
   559→              Import {selectedCandidates.size} candidate
   560→              {selectedCandidates.size === 1 ? "" : "s"}
   561→            </DialogTitle>
   562→            <DialogDescription>
   563→              Devices are created with status Unknown (or Unmanaged) until the
   564→              first successful poll. The credential profile is recorded for the
   565→              collector — secrets stay in the vault.
   566→            </DialogDescription>
   567→          </DialogHeader>
   568→          <div className="flex flex-col gap-4">

   651→          </DialogFooter>
