     1→/**
     2→ * Typed API client for the FayaNMS /api/v1 REST surface.
     3→ *
     4→ * Every client hook goes through `apiFetch` — components never call raw
     5→ * fetch. The standard envelope is unwrapped here and mapped to `ApiError`
     6→ * so callers only ever deal with data or a typed error.
     7→ */
     8→
     9→export class ApiError extends Error {
    10→  readonly code: string;
    11→  readonly status: number;
    12→
    13→  constructor(message: string, code: string, status: number) {
    14→    super(message);
    15→    this.name = "ApiError";
    16→    this.code = code;
    17→    this.status = status;
    18→  }
    19→}
    20→
    21→interface EnvelopeError {
    22→  success: false;
    23→  error: { code: string; message: string };
    24→}
    25→
    26→interface EnvelopeSuccess<T> {
    27→  success: true;
    28→  data: T;
    29→  meta?: Record<string, unknown>;
    30→}
    31→
    32→type Envelope<T> = EnvelopeSuccess<T> | EnvelopeError;
    33→
    34→/** GET/POST any /api/v1 endpoint; resolves with the full envelope (incl. meta). */
    35→export async function apiRequest<T>(
    36→  path: string,
    37→  init?: RequestInit
    38→): Promise<EnvelopeSuccess<T>> {
    39→  let response: Response;
    40→  try {
    41→    response = await fetch(path, {
    42→      cache: "no-store",
    43→      ...init,
    44→      headers: {
    45→        "Content-Type": "application/json",
    46→        ...(init?.headers ?? {}),
    47→      },
    48→    });
    49→  } catch {
    50→    throw new ApiError(
    51→      "Network request failed — the server could not be reached.",
    52→      "NETWORK_ERROR",
    53→      0
    54→    );
    55→  }
    56→
    57→  let envelope: Envelope<T> | null = null;
    58→  try {
    59→    envelope = (await response.json()) as Envelope<T>;
    60→  } catch {
    61→    envelope = null;
    62→  }
    63→
    64→  if (!envelope) {
    65→    throw new ApiError(
    66→      `Unexpected response from server (HTTP ${response.status}).`,
    67→      "BAD_ENVELOPE",
    68→      response.status
    69→    );
    70→  }
    71→
    72→  if (!envelope.success) {
    73→    throw new ApiError(envelope.error.message, envelope.error.code, response.status);
    74→  }
    75→  if (envelope.data === undefined) {
    76→    throw new ApiError(
    77→      "Response envelope is missing data.",
    78→      "BAD_ENVELOPE",
    79→      response.status
    80→    );
    81→  }
    82→
    83→  return envelope;
    84→}
    85→
    86→/** Convenience wrapper returning just `data`. */
    87→export async function apiFetch<T>(path: string, init?: RequestInit): Promise<T> {
    88→  const envelope = await apiRequest<T>(path, init);
    89→  return envelope.data;
    90→}

    92→/** Serialize a params object into a query string (skips empty values). */
    93→export function buildQueryString(
    94→  params: Record<string, unknown>
    95→): string {
    96→  const search = new URLSearchParams();
    97→  for (const [key, value] of Object.entries(params)) {
    98→    if (value === undefined || value === null || value === "") continue;
    99→    search.set(key, String(value));
   100→  }
   101→  const qs = search.toString();
   102→  return qs ? `?${qs}` : "";
   103→}
