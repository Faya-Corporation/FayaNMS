   175→        } catch {
   176→          // not a JSON envelope — keep the HTTP status message
   177→        }

   179→      }

   181→      const disposition = res.headers.get("Content-Disposition") ?? "";

   459→const policyFormSchema = z.object({
   460→  name: z.string().trim().min(1, "Name is required").max(120),
   461→  cronExpr: z
   462→    .string()
   463→    .trim()
   464→    .min(1, "Cron expression is required")
   465→    .refine(isValidCronExpr, {
   466→      message: "Use a 5-field numeric cron expression, e.g. 0 2 * * *",
   467→    }),
   468→  siteCodes: z.array(z.string()),
   469→  criticalities: z.array(z.string()),
   470→  statuses: z.array(z.string()),
   471→  // Kept as a string input so zodResolver's input/output types align with
   472→  // the form values; converted to a number on submit.
   473→  retentionDays: z
   474→    .string()
   475→    .trim()
   476→    .min(1, "Retention is required")
   477→    .regex(/^\d+$/, "Whole days only")
   478→    .refine((value) => {
   479→      const n = Number(value);
   480→      return n >= 1 && n <= 3650;
   481→    }, "Retention must be between 1 and 3650 days"),
   482→  isActive: z.boolean(),
   483→});
   484→
   485→type PolicyFormValues = z.infer<typeof policyFormSchema>;
   486→
   487→interface CheckboxGroupProps {
   488→  label: string;
   489→  hint?: string;
   490→  options: { value: string; label: string }[];
   491→  value: string[];
   492→  onChange: (next: string[]) => void;
   493→  error?: string;
   494→}
   495→
   496→function CheckboxGroup({
   497→  label,
   498→  hint,
   499→  options,
   500→  value,
   501→  onChange,
   502→  error,
   503→}: CheckboxGroupProps) {
   504→  return (
   505→    <div className="flex flex-col gap-1.5">
   506→      <Label>{label}</Label>
   507→      <div className="grid grid-cols-2 gap-2 rounded-md border bg-surface-subtle p-3">
   508→        {options.map((option) => {
   509→          const checked = value.includes(option.value);
   510→          return (
   511→            <label
   512→              className="flex min-h-11 cursor-pointer items-center gap-2 text-sm md:min-h-0"
   513→              key={option.value}
   514→            >
   515→              <Checkbox
   516→                aria-label={`Scope ${label}: ${option.label}`}
   517→                checked={checked}
   518→                onCheckedChange={(checkedState) =>
   519→                  onChange(
   520→                    checkedState === true
   521→                      ? [...value, option.value]
   522→                      : value.filter((entry) => entry !== option.value)
   523→                  )
   524→                }
   525→              />
   526→              {option.label}
   527→            </label>
   528→          );
   529→        })}
   530→      </div>
   531→      {hint && <p className="text-xs text-muted-foreground">{hint}</p>}
   532→      {error && <p className="text-xs text-danger">{error}</p>}
   533→    </div>
   534→  );
   535→}

   537→function PolicyFormDialog({
   538→  open,
   539→  onOpenChange,
   540→  policy,
   541→}: {
   542→  open: boolean;
   543→  onOpenChange: (open: boolean) => void;
   544→  /** When provided the dialog edits this policy instead of creating one. */
   545→  policy?: BackupPolicyRow | null;
   546→}) {
   547→  const meta = useMeta();
   548→  const createPolicy = useCreateBackupPolicy();
   549→  const updatePolicy = useUpdateBackupPolicy();
   550→
   551→  const editing = Boolean(policy);
   552→  const defaultValues: PolicyFormValues = useMemo(
   553→    () => ({
   554→      name: policy?.name ?? "",
   555→      cronExpr: policy?.cronExpr ?? "0 2 * * *",
   556→      siteCodes: policy?.scope.siteCodes ?? [],
   557→      criticalities: policy?.scope.criticalities ?? [],
   558→      statuses: policy?.scope.statuses ?? [],
   559→      retentionDays: String(policy?.retentionDays ?? 90),
   560→      isActive: policy?.isActive ?? true,
   561→    }),
   562→    [policy]
   563→  );
   564→
   565→  const form = useForm<PolicyFormValues>({
   566→    defaultValues,
   567→    resolver: zodResolver(policyFormSchema),
   568→    mode: "onSubmit",
   569→  });
   570→
   571→  // useWatch (compiler-safe) instead of form.watch() — see device-form-sheet.
   572→  const cronExpr = useWatch({ control: form.control, name: "cronExpr" });
   573→  const siteCodes = useWatch({ control: form.control, name: "siteCodes" });
   574→  const criticalities = useWatch({ control: form.control, name: "criticalities" });
   575→  const statuses = useWatch({ control: form.control, name: "statuses" });
   576→  const isActive = useWatch({ control: form.control, name: "isActive" });
   577→
   578→  useEffect(() => {
   579→    if (open) {
   580→      form.reset(defaultValues);
   581→    }
   582→  }, [open, defaultValues, form]);
   583→
   584→  const pending = createPolicy.isPending || updatePolicy.isPending;
   585→
   586→  const onSubmit = (values: PolicyFormValues) => {
   587→    const payload = {
   588→      name: values.name,
   589→      cronExpr: values.cronExpr,
   590→      scope: {
   591→        siteCodes: values.siteCodes,
   592→        criticalities: values.criticalities,
   593→        statuses: values.statuses,
   594→      },
   595→      retentionDays: Number(values.retentionDays),
   596→      isActive: values.isActive,
   597→    };
   598→    if (editing && policy) {
   599→      updatePolicy.mutate(
   600→        { id: policy.id, data: payload },
   601→        { onSuccess: () => onOpenChange(false) }
   602→      );
   603→      return;
   604→    }
   605→    createPolicy.mutate(payload, { onSuccess: () => onOpenChange(false) });
   606→  };
   607→
   608→  const sites = meta.data?.sites ?? [];
   609→  const siteOptions = [
   610→    { value: "*", label: "All sites (fleet-wide)" },
   611→    ...sites.map((site) => ({ value: site.code, label: `${site.name} (${site.code})` })),
   612→  ];
   613→
   614→  /** "All sites" is mutually exclusive with explicit site codes. */
   615→  const handleSiteCodesChange = (next: string[]) => {
   616→    const hadStar = siteCodes.includes("*");
   617→    const hasStar = next.includes("*");
   618→    if (!hadStar && hasStar) {
   619→      form.setValue("siteCodes", ["*"], { shouldValidate: true });
   620→      return;
   621→    }
   622→    form.setValue(
   623→      "siteCodes",
   624→      next.filter((code) => code !== "*" || next.length === 1),
   625→      { shouldValidate: true }
   626→    );
   627→  };
   628→
   629→  const hint = cronHint(cronExpr ?? "");
   630→
   631→  return (
   632→    <Dialog onOpenChange={onOpenChange} open={open}>
   633→      <DialogContent className="max-h-[90vh] overflow-y-auto sm:max-w-lg">
   634→        <DialogHeader>
   635→          <DialogTitle>{editing ? "Edit backup policy" : "New backup policy"}</DialogTitle>
   636→          <DialogDescription>
   637→            The worker scheduler evaluates active policies every 30 seconds —
   638→            saved changes become live schedules automatically.
   639→          </DialogDescription>
   640→        </DialogHeader>
   641→
   642→        <form className="flex flex-col gap-4" onSubmit={form.handleSubmit(onSubmit)}>
   643→          <div className="flex flex-col gap-1.5">
   644→            <Label htmlFor="policy-name">Name *</Label>
   645→            <Input
   646→              {...form.register("name")}
   647→              aria-invalid={Boolean(form.formState.errors.name)}
   648→              id="policy-name"
   649→              placeholder="Daily Full Fleet 02:00"
   650→            />
   651→            {form.formState.errors.name && (
   652→              <p className="text-xs text-danger">{form.formState.errors.name.message}</p>
   653→            )}
   654→          </div>
   655→
   656→          <div className="flex flex-col gap-1.5">
   657→            <Label htmlFor="policy-cron">Schedule (cron) *</Label>
   658→            <div className="flex flex-wrap gap-1.5">
   659→              {CRON_PRESETS.map((preset) => (
   660→                <Button
   661→                  key={preset.value}
   662→                  onClick={() =>
   663→                    form.setValue("cronExpr", preset.value, { shouldValidate: true })
   664→                  }
   665→                  size="sm"
   666→                  type="button"
   667→                  variant={cronExpr === preset.value ? "secondary" : "outline"}
   668→                >
   669→                  {preset.label}
   670→                </Button>
   671→              ))}
   672→            </div>
   673→            <Input
   674→              {...form.register("cronExpr")}
   675→              aria-invalid={Boolean(form.formState.errors.cronExpr)}
   676→              className="font-tech ltr-technical"
   677→              id="policy-cron"
   678→              placeholder="0 2 * * *"
   679→            />
   680→            {hint && !form.formState.errors.cronExpr && (
   681→              <p className="text-xs text-info">{hint}</p>
   682→            )}
   683→            {form.formState.errors.cronExpr && (
   684→              <p className="text-xs text-danger">{form.formState.errors.cronExpr.message}</p>
   685→            )}
   686→          </div>
   687→
   688→          <CheckboxGroup
   689→            error={form.formState.errors.siteCodes?.message}
   690→            hint="Devices from the selected sites. Leave empty for all sites."
   691→            label="Sites"
   692→            onChange={handleSiteCodesChange}
   693→            options={siteOptions}
   694→            value={siteCodes ?? []}
   695→          />
   696→
   697→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   698→            <CheckboxGroup
   699→              label="Criticalities"
   700→              onChange={(next) =>
   701→                form.setValue("criticalities", next, { shouldValidate: true })
   702→              }
   703→              options={CRITICALITY_OPTIONS}
   704→              value={criticalities ?? []}
   705→            />
   706→            <CheckboxGroup
   707→              hint="Offline and unmanaged devices are never scheduled."
   708→              label="Statuses"
   709→              onChange={(next) =>
   710→                form.setValue("statuses", next, { shouldValidate: true })
   711→              }
   712→              options={STATUS_OPTIONS}
   713→              value={statuses ?? []}
   714→            />
   715→          </div>
   716→
   717→          <div className="grid grid-cols-1 gap-4 sm:grid-cols-2">
   718→            <div className="flex flex-col gap-1.5">
   719→              <Label htmlFor="policy-retention">Retention (days) *</Label>
   720→              <Input
   721→                {...form.register("retentionDays")}
   722→                aria-invalid={Boolean(form.formState.errors.retentionDays)}
   723→                id="policy-retention"
   724→                inputMode="numeric"
   725→              />
   726→              <p className="text-xs text-muted-foreground">
   727→                Historical versions older than this are pruned by the scheduler.
   728→              </p>
   729→              {form.formState.errors.retentionDays && (
   730→                <p className="text-xs text-danger">
   731→                  {form.formState.errors.retentionDays.message}
   732→                </p>
   733→              )}
   734→            </div>
   735→            <div className="flex flex-col gap-1.5">
   736→              <Label htmlFor="policy-active">Active</Label>
   737→              <div className="flex h-9 items-center gap-2">
   738→                <Switch
   739→                  aria-label="Policy active"
   740→                  checked={isActive}
   741→                  id="policy-active"
   742→                  onCheckedChange={(checked) => form.setValue("isActive", checked)}
   743→                />
   744→                <span className="text-sm text-muted-foreground">
   745→                  {isActive ? "Scheduler runs this policy" : "Paused"}
   746→                </span>
   747→              </div>
   748→            </div>
   749→          </div>
   750→
   751→          <DialogFooter>
   752→            <Button
   753→              disabled={pending}
   754→              onClick={() => onOpenChange(false)}
   755→              type="button"
   756→              variant="outline"
   757→            >
   758→              Cancel
   759→            </Button>
   760→            <Button disabled={pending} type="submit">
   761→              {pending && <LoaderCircle aria-hidden="true" className="animate-spin" />}
   762→              {editing ? "Save changes" : "Create policy"}
   763→            </Button>
   764→          </DialogFooter>
   765→        </form>
   766→      </DialogContent>
   767→    </Dialog>
   768→  );

   775→function ScopeChips({ policy }: { policy: BackupPolicyRow }) {
   776→  const { scope } = policy;
   777→  const chips: { label: string; value: string }[] = [];
   778→
   779→  if (scope.siteCodes.length > 0) {
   780→    chips.push({
   781→      label: "Sites",
   782→      value: scope.siteCodes.includes("*")
   783→        ? "All sites"
   784→        : scope.siteCodes.join(", "),
   785→    });
   786→  }
   787→  for (const criticality of scope.criticalities) {
   788→    chips.push({
   789→      label: "Criticality",
   790→      value:
   791→        CRITICALITY_OPTIONS.find((option) => option.value === criticality)?.label ??
   792→        criticality,
   793→    });
   794→  }
   795→  for (const status of scope.statuses) {
   796→    chips.push({
   797→      label: "Status",
   798→      value: getStatusConfig(DEVICE_STATUS, status).label,
   799→    });
   800→  }
   801→
   802→  if (chips.length === 0) {
   803→    return (
   804→      <span className="text-xs text-muted-foreground">
   805→        Fleet-wide — all manageable devices
   806→      </span>
   807→    );
   808→  }
   809→
   810→  return (
   811→    <div className="flex flex-wrap gap-1">
   812→      {chips.map((chip) => (
   813→        <FilterChip key={`${chip.label}-${chip.value}`} label={chip.label} value={chip.value} />
   814→      ))}
   815→    </div>
   816→  );
   817→}
